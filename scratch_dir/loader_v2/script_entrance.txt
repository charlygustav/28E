            if (typeof gsap !== 'undefined') {

                gsap.set('.loader-center', { opacity: 1 });

                gsap.set('.loader-track', { opacity: 0, scaleX: 0 });

                // Build entrance timeline
                entranceTl
                    // Phase 1: Ambient atmosphere
                    .to('.loader-ambient', { opacity: 1, scale: 1, duration: 1.5, ease: 'power2.inOut' }, 0)
                    // Phase 2: Letters reveal one by one
                    .to('.loader-char', {
                        opacity: 1, y: 0, scale: 1,
                        duration: 0.7, stagger: 0.12, ease: 'back.out(1.7)'
                    }, 0.3)
                    // Phase 3: Subtitle
                    .to('.loader-sub', { opacity: 1, y: 0, duration: 0.5 }, 0.85)
                    // Phase 4: Progress track expands from center
                    .to('.loader-track', { opacity: 1, scaleX: 1, duration: 0.6, ease: 'power2.out' }, 0.95)
                    // Phase 5: Bottom percentage
                    .to('.loader-bottom', { opacity: 1, y: 0, duration: 0.5 }, 0.95);

                // ── Continuous Animations ──
                // Ambient glow breathing
                gsap.to('.loader-ambient', {
                    scale: 1.2, opacity: 0.7, duration: 3, repeat: -1, yoyo: true, ease: 'sine.inOut'
                });

                            duration: gsap.utils.random(0.8, 1.5),
                            delay: gsap.utils.random(0.8, 2.5), ease: 'power2.out'
                        });
                        gsap.to(p, {

                            duration: gsap.utils.random(5, 10), repeat: -1, yoyo: true,
                            ease: 'sine.inOut', delay: gsap.utils.random(0, 3)
                        });
                    }
